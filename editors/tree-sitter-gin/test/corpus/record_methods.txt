    values Buffer(Byte)

    --- Copies the input elements.
    from_values(source [in Byte; count]) Self:
        values: Buffer.with_capacity(Byte, count as Length)
        for index in 0 to count
            values.push(source.[index] as Byte)
        return Self(values)

    copy(self ref) Self: return Self(values: self.values.copy)

    #infix('+')
    merge(self own, other own Self) Self:
        Packet(values) := self
        Packet(values: right) := other
        values.append(right)
        return Self(values)

    append(self mut, other own Self): return self.values.append(other.values)
